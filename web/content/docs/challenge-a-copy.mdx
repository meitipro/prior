On the site, open your revealed entry, press **Challenge a later copy of this**, pick the later entry, read the two texts side by side, and sign. Only the author of the earlier entry can file, with a stake of exactly 5 GEN, and a pair can be challenged once.

<Steps>
<Step>

**Stake.** 5 GEN, sent with `challenge`.

</Step>
<Step>

**Read.** Anyone calls `judge`. The validators see two texts and no dates or names, and answer SUBSTANTIAL, PARTIAL or NONE.

</Step>
<Step>

**Mark.** The contract derives the verdict from the label and the seal dates and credits balances by rule.

</Step>
<Step>

**Withdraw.** The only call that pays is `withdraw`, in its own transaction.

</Step>
</Steps>

import { createClient } from 'genlayer-js';
import { studioDevnet } from 'genlayer-js/chains';

const PRIOR = '0xCc03C597dE98A0ecDEFBd500bbCF6314fCf03da4';
const chain = { ...studioDevnet, id: 61997, rpcUrls: { default: { http: ['https://studio-next.genlayer.com/api'] } } };

const client = createClient({ chain });
const [id = '1'] = process.argv.slice(2);
const e = JSON.parse(await client.readContract({ address: PRIOR, functionName: 'get_entry', args: [Number(id)] }));

if (!e.found) throw new Error('no such entry');
console.log(`#${e.id} ${e.state}, sealed ${new Date(e.sealed_at * 1000).toISOString()}, seal ${e.seal.slice(0, 12)}…`);
if (e.revealed) console.log(`  ${e.text}`);
console.log(`  overlaps #${e.overlaps || 'none'}, ${e.related} related, ${e.cleared} cleared`);

<Tab value="Python">

```python title="web/examples/read_entry.py"


import json
import sys

from genlayer_py import create_account, create_client
from genlayer_py.chains import studio_devnet

PRIOR = "0xCc03C597dE98A0ecDEFBd500bbCF6314fCf03da4"
studio_devnet.rpc_urls = {"default": {"http": ["https://studio-next.genlayer.com/api"]}}

reader = create_account()
client = create_client(chain=studio_devnet, account=reader)
entry_id = int(sys.argv[1] if len(sys.argv) > 1 else 1)
e = json.loads(client.read_contract(address=PRIOR, function_name="get_entry", args=[entry_id], account=reader))

if not e["found"]:
    raise SystemExit("no such entry")
print(json.dumps({
    "state": e["state"],
    "sealed_at": e["sealed_at"],
    "bond_gen": int(e["bond"]) / 10**18,
    "text": e["text"],
    "overlaps": e["overlaps"],
    "related": e["related"],
    "cleared": e["cleared"],
}, ensure_ascii=False))
```
